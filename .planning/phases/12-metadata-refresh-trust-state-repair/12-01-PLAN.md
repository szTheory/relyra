def build_candidate(parsed) do
  certificate_pems = Enum.map(Map.fetch!(parsed, :certificates), &to_pem/1)
  certificate_fingerprints = Enum.map(certificate_pems, &sha256/1)
  certificate_facts = Enum.map(certificate_pems, &certificate_fact(&1))
  %Candidate{certificate_facts: certificate_facts, certificate_pems: certificate_pems, certificate_fingerprints: certificate_fingerprints}
end
