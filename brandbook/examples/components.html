case Relyra.consume_response(conn, raw_response, opts) do
  {:ok, %Relyra.Assertion{subject: subject}} ->
    {:ok, subject}

  {:error, %Relyra.Rejection{reason: reason}} ->
    # Typed rejection — never a silent compromise.
    Logger.warning("SAML rejected: #{inspect(reason)}")
    {:error, reason}
end
